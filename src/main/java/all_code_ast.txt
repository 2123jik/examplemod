package com.example.examplemod.client;
@EventBusSubscriber(modid = ExampleMod.MODID, value = Dist.CLIENT)
p c ChestScreenOverlay {
    prs f int OVERLAY_Z = 300;
    @SubscribeEvent
    ps v onRenderGuiForeground(ContainerScreenEvent.Render.Foreground event) {
        var guiGraphics = event.getGuiGraphics();
        var screen = event.getContainerScreen();
        for (Slot slot : screen.getMenu().slots) {
            if (!slot.isActive() || !slot.hasItem())
                continue;
            ItemStack stack = slot.getItem();
            int x = slot.x;
            int y = slot.y;
            var rarityHolder = stack.get(Apoth.Components.RARITY);
            if (rarityHolder != null && rarityHolder.isBound()) {
                renderPip(guiGraphics, x + 1, y + 1, rarityHolder.get().color().getValue() | 0xFF000000);
            }
            if (stack.getItem() instanceof Scroll) {
                Integer scrollColor = getScrollRarityColor(stack);
                if (scrollColor != null)
                    renderPip(guiGraphics, x + 1, y + 1, scrollColor);
            }
            Purity purity = stack.get(Apoth.Components.PURITY);
            if (purity != null && purity.getColor() != null) {
                int size = 3;
                renderPip(guiGraphics, slot.x + 16 - size - 1, y + 1, purity.getColor().getValue() | 0xFF000000);
            }
        }
    }
    prs Integer getScrollRarityColor(ItemStack stack) {
        CustomData customData = stack.get(DataComponents.CUSTOM_DATA);
        if (customData != null) {
            CompoundTag tag = customData.getUnsafe();
            if (tag.getBoolean("is_unidentified") && tag.contains("spell_rarity")) {
                try {
                    SpellRarity rarity = SpellRarity.valueOf(tag.getString("spell_rarity").toUpperCase(Locale.ROOT));
                    Integer color = rarity.getChatFormatting().getColor();
                    r color != null ? (color | 0xFF000000) : null;
                } cat (Exception ignored) {
                }
            }
        }
        if (ISpellContainer.isSpellContainer(stack)) {
            ISpellContainer container = ISpellContainer.get(stack);
            if (container != null && !container.isEmpty()) {
                SpellData spellData = container.getSpellAtIndex(0);
                if (spellData != null && spellData != SpellData.EMPTY) {
                    Integer color = spellData.getRarity().getChatFormatting().getColor();
                    r color != null ? (color | 0xFF000000) : null;
                }
            }
        }
        r null;
    }
    prs v renderPip(GuiGraphics guiGraphics, int x, int y, int argbColor) {
        int size = 3;
        guiGraphics.fill(x - 1, y - 1, x + size + 1, y + size + 1, OVERLAY_Z - 1, 0x60000000);
        guiGraphics.fill(x, y, x + size, y + size, OVERLAY_Z, argbColor);
    }
}

package com.example.examplemod.client;
@EventBusSubscriber(modid = ExampleMod.MODID, value = Dist.CLIENT)
p c ItemCopyHandler {
    @SubscribeEvent
    ps v onScreenKeyPressed(ScreenEvent.KeyPressed.Pre event) {
        if (event.getKeyCode() == GLFW.GLFW_KEY_C && Screen.hasControlDown()) {
            if (event.getScreen() instanceof AbstractContainerScreen<?> containerScreen) {
                Slot slot = containerScreen.getSlotUnderMouse();
                if (slot != null && slot.hasItem()) {
                    copyItemId(slot.getItem());
                    event.setCanceled(T);
                }
            }
        }
    }
    @SubscribeEvent
    ps v onGameKeyPressed(InputEvent.Key event) {
        if (event.getAction() == GLFW.GLFW_PRESS && event.getKey() == GLFW.GLFW_KEY_C && Screen.hasControlDown()) {
            Minecraft mc = Minecraft.getInstance();
            if (mc.screen == null && mc.player != null) {
                ItemStack mainHand = mc.player.getMainHandItem();
                if (!mainHand.isEmpty()) {
                    copyItemId(mainHand);
                }
            }
        }
    }
    prs v copyItemId(ItemStack stack) {
        ResourceLocation id = BuiltInRegistries.ITEM.getKey(stack.getItem());
        Str idStr = id.toString();
        Minecraft.getInstance().keyboardHandler.setClipboard(idStr);
        Player player = Minecraft.getInstance().player;
        if (player != null) {
            player.displayClientMessage(Component.literal("§a[已复制 ID] §f" + idStr), T);
        }
    }
}

package com.example.examplemod.client;
p c StatusHudRenderer {
    prs f int START_X = 12;
    prs f int START_Y = 12;
    prs f int BAR_WIDTH = 110;
    prs f int BAR_HEIGHT = 10;
    prs f int BAR_SPACING = 14;
    prs f float CORNER_RADIUS = 3.5f;
    ps v render(GuiGraphics gr, float partialTick) {
        Minecraft mc = Minecraft.getInstance();
        Player player = mc.player;
        if (player == null || mc.options.hideGui || player.isSpectator()) {
            r;
        }
        ExtendedGuiGraphics exGr = n ExtendedGuiGraphics(gr);
        Font font = mc.font;
        int currentY = START_Y;
        float health = player.getHealth();
        float maxHealth = player.getMaxHealth();
        float absorption = player.getAbsorptionAmount();
        float totalHealthRatio = Mth.clamp(health / maxHealth, 0.0f, 1.0f);
        drawBarBackground(exGr, START_X, currentY, BAR_WIDTH, BAR_HEIGHT);
        int fillWidth = (int) (BAR_WIDTH * totalHealthRatio);
        if (fillWidth > 0) {
            exGr.setGradient(ExtendedGuiGraphics.Orientation.LEFT_RIGHT, 0xFFD32F2F, 0xFFFF5252);
            exGr.fillRoundRect(START_X, currentY, START_X + fillWidth, currentY + BAR_HEIGHT, CORNER_RADIUS);
        }
        if (absorption > 0) {
            float absRatio = Mth.clamp(absorption / maxHealth, 0.0f, 1.0f);
            int absWidth = (int) (BAR_WIDTH * absRatio);
            exGr.setColor(0xFFFFD700);
            exGr.setStrokeWidth(1.2f);
            exGr.strokeRoundRect(START_X, currentY, START_X + absWidth, currentY + BAR_HEIGHT, CORNER_RADIUS);
        }
        drawBarBorder(exGr, START_X, currentY, BAR_WIDTH, BAR_HEIGHT);
        Str hpText = Str.format("%.1f", health + absorption);
        gr.drawString(font, hpText, START_X + BAR_WIDTH + 6, currentY + 1, 0xFFFFFF, T);
        currentY += BAR_SPACING;
        FoodData foodData = player.getFoodData();
        int foodLevel = foodData.getFoodLevel();
        float saturation = foodData.getSaturationLevel();
        float foodRatio = Mth.clamp(foodLevel / 20.0f, 0.0f, 1.0f);
        drawBarBackground(exGr, START_X, currentY, BAR_WIDTH, BAR_HEIGHT);
        int foodFillWidth = (int) (BAR_WIDTH * foodRatio);
        if (foodFillWidth > 0) {
            exGr.setGradient(ExtendedGuiGraphics.Orientation.LEFT_RIGHT, 0xFFE65100, 0xFFFFB74D);
            exGr.fillRoundRect(START_X, currentY, START_X + foodFillWidth, currentY + BAR_HEIGHT, CORNER_RADIUS);
        }
        float satRatio = Mth.clamp(saturation / 20.0f, 0.0f, 1.0f);
        int satFillWidth = (int) (BAR_WIDTH * satRatio);
        if (satFillWidth > 0) {
            exGr.setColor(0xAAFFF176);
            exGr.fillRoundRect(START_X + 1, currentY + 1, START_X + satFillWidth - 1, currentY + 3, 1.0f);
        }
        drawBarBorder(exGr, START_X, currentY, BAR_WIDTH, BAR_HEIGHT);
        Str foodText = Str.valueOf(foodLevel);
        gr.drawString(font, foodText, START_X + BAR_WIDTH + 6, currentY + 1, 0xFFE0E0E0, T);
    }
    prs v drawBarBackground(ExtendedGuiGraphics exGr, int x, int y, int w, int h) {
        exGr.setColor(0x88000000);
        exGr.fillRoundRect(x, y, x + w, y + h, CORNER_RADIUS);
    }
    prs v drawBarBorder(ExtendedGuiGraphics exGr, int x, int y, int w, int h) {
        exGr.setColor(0x33FFFFFF);
        exGr.setStrokeWidth(1.0f);
        exGr.strokeRoundRect(x, y, x + w, y + h, CORNER_RADIUS);
    }
}

                provider.appendContributions(context, contributions);
            }
        }
        if (Config.SHOW_CUSTOM_RULES.get()) {
            TooltipRules.add(stack, contributions);
        }
        TooltipContributors.contribute(context, contributions);
        event.getTooltipElements().addAll(TooltipUtil.buildElements(contributions));
        List<Component> lines = n ArrayList<>(event.getTooltipElements().size());
        for (var element : event.getTooltipElements()) {
            element.ifLeft(text -> lines.add(text instanceof Component component ? component.copy() : Component.literal(text.getString())));

pf c TooltipUtil {
    prs f Comparator<TooltipContribution> ORDER = Comparator.comparingInt((TooltipContribution c) -> c.category().order()).thenComparingInt(TooltipContribution::priority).thenComparing(TooltipContribution::key);
    pr TooltipUtil() {
    }
    ps List<Either<FormattedText, TooltipComponent>> buildElements(List<TooltipContribution> contributions) {
        int limit = Math.max(0, Config.MAX_LINES.get());
        if (limit == 0 || contributions.isEmpty())
            r List.of();
        List<TooltipContribution> visible = n ArrayList<>(contributions.size());
        for (TooltipContribution contribution : contributions) {
            if (contribution.isVisible())
                visible.add(contribution);
        }
        visible.sort(ORDER);
        List<Either<FormattedText, TooltipComponent>> result = n ArrayList<>(Math.min(limit, visible.size()));
        Set<Str> seen = n HashSet<>();
        for (TooltipContribution contribution : visible) {
            if (!seen.add(contribution.key()))
                continue;
            result.add(contribution.content().mapBoth(text -> (FormattedText) text, visual -> visual));
            if (result.size() == limit)
                break;
        }
        r result;
    }
    ps float tickRate(TooltipContext context) {
        float rate = context.vanillaContext() == null ? 20.0F : context.vanillaContext().tickRate();
        r Float.isFinite(rate) && rate > 0.0F ? rate : 20.0F;
    }
    ps float unit(float value) {
        r Float.isFinite(value) ? Mth.clamp(value, 0.0F, 1.0F) : 0.0F;
    }
    ps float ratio(int value, int maximum) {
        r maximum > 0 ? unit((float) value / maximum) : 0.0F;
    }
    ps List<ItemStack> copyStacks(List<ItemStack> stacks) {
        Objects.requireNonNull(stacks, "stacks");
        r stacks.stream().map(ItemStack::copy).toList();
    }
    ps int lerpColor(int from, int to, float factor) {
        float t = unit(factor);
        int r = Math.round(Mth.lerp(t, from >> 16 & 255, to >> 16 & 255));
        int g = Math.round(Mth.lerp(t, from >> 8 & 255, to >> 8 & 255));
        int b = Math.round(Mth.lerp(t, from & 255, to & 255));
        r 0xFF000000 | r << 16 | g << 8 | b;
    }
}

            Level level = Minecraft.getInstance().level;
            if (level == null)
                r F;
            Map<Str, ISlotType> slotTypes = CuriosApi.getItemStackSlots(stack, level);
            r slotTypes != null && !slotTypes.isEmpty();
        }
        prs bool tryEquip(Player player, ItemStack stack) {
            Level level = player.level();
            Map<Str, ISlotType> slotTypes = CuriosApi.getItemStackSlots(stack, level);
            if (slotTypes == null || slotTypes.isEmpty())
                r F;
            var curiosOpt = CuriosApi.getCuriosInventory(player);
            if (curiosOpt.isEmpty())
                r F;
            ICuriosItemHandler handler = curiosOpt.get();
            handler.getCurios().forEach((slotId, stacksHandler) -> {
                IDynamicStackHandler stacks = stacksHandler.getStacks();
                for (int i = 0; i < stacks.getSlots(); i++) {
                    stacks.setStackInSlot(i, ItemStack.EMPTY);

                for (int i = 0; i < cosmeticStacks.getSlots(); i++) {

package com.example.examplemod.compat.l2hostility;
p c SpellTraitHelper {
    ps int getDynamicColor(AbstractSpell spell) {
        if (spell == null)
            r 0xFFFFFF;
        SchoolType school = spell.getSchoolType();
        if (school == null)
            r 0xFFFFFF;
        Vector3f c = school.getTargetingColor();
        if (c == null)
            r 0xFFFFFF;
        int r = Mth.clamp((int) (c.x() * 255.0F), 0, 255);
        int g = Mth.clamp((int) (c.y() * 255.0F), 0, 255);
        int b = Mth.clamp((int) (c.z() * 255.0F), 0, 255);
        int lum = (r * 299 + g * 587 + b * 114) / 1000;
        if (lum < 60) {
            r = Math.min(255, r + 80);
            g = Math.min(255, g + 80);
            b = Math.min(255, b + 80);
        }
        r (r << 16) | (g << 8) | b;
    }
    ps int scaleSpellLevel(int traitLevel, int maxTraitLevel, AbstractSpell spell) {
        if (spell == null)
            r 1;
        int minS = spell.getMinLevel();
        int maxS = spell.getMaxLevel();
        if (minS >= maxS)
            r minS;
        int maxT = Math.max(1, maxTraitLevel);
        if (maxT == 1)
            r minS;
        float progress = (float) (traitLevel - 1) / (float) (maxT - 1);
        int result = Math.round(minS + progress * (maxS - minS));
        r Mth.clamp(result, minS, maxS);
    }
    ps int getSafeCooldown(AbstractSpell spell) {
        if (spell == null)
            r 60;
        int nativeCd = spell.getSpellCooldown();
        int castTime = spell.getCastTime(spell.getMinLevel());
        int floor = castTime + 40;
        r Math.max(nativeCd, Math.max(floor, 20));
    }
}

        if (!level.isClientSide && level instanceof ServerLevel serverLevel) {
            tickSmelting(serverLevel);
        }
        sup.broadcastChanges();
    }
    pr v tickSmelting(ServerLevel level) {
        ItemStack input = container.getItem(0);
        ItemStack fuel = container.getItem(1);
        ItemStack output = container.getItem(2);
        int litTime = data.get(0);
        int litDuration = data.get(1);
        int cookingProgress = data.get(2);
        int cookingTotal = data.get(3) <= 0 ? 200 : data.get(3);
        bool isLit = litTime > 0;
        if (isLit) {
            litTime--;
            data.set(0, litTime);
        }
        if (!input.isEmpty()) {
            SingleRecipeInput recipeInput = n SingleRecipeInput(input);
            var recipeHolder = level.getRecipeManager().getRecipeFor(RecipeType.SMELTING, recipeInput, level).orElse(null);
            if (recipeHolder != null) {
                ItemStack result = recipeHolder.value().assemble(recipeInput, level.registryAccess());
                bool canInsert = output.isEmpty() || (ItemStack.isSameItemSameComponents(output, result) && output.getCount() + result.getCount() <= output.getMaxStackSize());
                if (canInsert) {
                    if (litTime <= 0 && !fuel.isEmpty()) {
                        int burn = fuel.getBurnTime(RecipeType.SMELTING);
                        if (burn > 0) {
                            litTime = burn;
                            litDuration = burn;
                            data.set(0, litTime);
                            data.set(1, litDuration);
                            ItemStack remainder = fuel.getCraftingRemainingItem();
                            fuel.shrink(1);
                            if (fuel.isEmpty() && !remainder.isEmpty()) {
                                container.setItem(1, remainder);
                            }
                            container.setChanged();
                        }
                    }
                    if (litTime > 0) {
                        cookingProgress++;
                        if (cookingProgress >= cookingTotal) {
                            cookingProgress = 0;
                            input.shrink(1);
                            if (output.isEmpty()) {
                                container.setItem(2, result.copy());
                            } el {
                                output.grow(result.getCount());
                            }
                            container.setChanged();
                        }
                        data.set(2, cookingProgress);
                    }
                } el {
                    data.set(2, 0);
                }
            } el {
                data.set(2, 0);
            }
        } el {
            data.set(2, 0);
        }
    }
    p v removed(Player player) {
        sup.removed(player);
        if (onClose != null) {
            onClose.run();
        }
    }
}

    @SubscribeEvent
    ps v onEntityJoinLevel(EntityJoinLevelEvent event) {

            r;
        }

            applyHealthModifier(player, F);
        }
    }
    @SubscribeEvent
    ps v onPlayerRespawn(PlayerEvent.PlayerRespawnEvent event) {
        if (event.getEntity() instanceof ServerPlayer player) {
            bool isDeathRespawn = !event.isEndConquered();
            applyHealthModifier(player, isDeathRespawn);
            if (isDeathRespawn) {
                player.setHealth(player.getMaxHealth());
            }
        }
    }
    ps v applyHealthModifier(ServerPlayer player, bool forceFullHealth) {
        AttributeInstance maxHealthAttr = player.getAttribute(Attributes.MAX_HEALTH);
        if (maxHealthAttr == null) {
            r;
        }
        AttributeModifier existing = maxHealthAttr.getModifier(HEALTH_MULTIPLIER_ID);
        if (existing != null && existing.amount() == MULTIPLIER_AMOUNT) {
            r;
        }
        float oldMaxHealth = player.getMaxHealth();
        float currentHealth = player.getHealth();
        maxHealthAttr.addOrReplacePermanentModifier(HEALTH_MODIFIER);
        float newMaxHealth = player.getMaxHealth();
        if (!player.isDeadOrDying() && currentHealth > 0.0F && oldMaxHealth > 0.0F) {
            if (forceFullHealth) {
                player.setHealth(newMaxHealth);
            } el if (oldMaxHealth != newMaxHealth) {
                float ratio = currentHealth / oldMaxHealth;
                float adjustedHealth = Mth.clamp(ratio * newMaxHealth, 1.0F, newMaxHealth);
                player.setHealth(adjustedHealth);
            }
        }
    }
}

    p v shuffleAndSplitItems(ObjectArrayList<ItemStack> stacks, int emptySlotsCount, RandomSource random) {
        Map<ItemStackKey, List<ItemStack>> groups = n LinkedHashMap<>(stacks.size());
        for (ItemStack s : stacks) {
            if (s.isEmpty())
                continue;
            groups.computeIfAbsent(n ItemStackKey(s), k -> n ArrayList<>()).add(s);
        }
        for (List<ItemStack> group : groups.values()) {
            if (group.size() < 2)
                continue;
            int maxStack = group.get(0).getMaxStackSize();
            int total = 0;
            for (ItemStack s : group) total += s.getCount();
            int idx = 0;
            wh (total > 0 && idx < group.size()) {
                int give = Math.min(maxStack, total);
                group.get(idx).setCount(give);
                total -= give;
                idx++;
            }
            for (; idx < group.size(); idx++) group.get(idx).setCount(0);
        }
    }

package com.example.examplemod.util;
p c ArtifactStatFormatter {
    ps Component format(Component originalText, StatEntry entry, int level, bool isMain) {
        if (originalText == null)
            r null;
        StatType type = (StatType) entry.type().value();
        double baseVal = type.getBaseValue();
        double multiplier = baseVal > 0 ? (entry.value() / baseVal) : entry.value();
        double percent;
        if (isMain) {
            double maxTheoryMultiplier = type.base_high() + level * type.main_high();
            if (maxTheoryMultiplier <= 0)
                r originalText;
            percent = (multiplier / maxTheoryMultiplier) * 100.0;
        } el {
            double singleMax = type.base_high();
            if (singleMax <= 0)
                r originalText;
            int rolls = Math.max(1, (int) Math.ceil((multiplier - 1e-5) / singleMax));
            double maxTheoryMultiplier = rolls * singleMax;
            percent = (multiplier / maxTheoryMultiplier) * 100.0;
        }
        percent = Math.min(100.0, Math.max(0.0, percent));
        ChatFormatting color;
        bool isPerfect = F;
        if (percent >= 99.95) {
            color = ChatFormatting.GOLD;
            isPerfect = T;
        } el if (percent >= 90.0) {
            color = ChatFormatting.YELLOW;
        } el if (percent >= 75.0) {
            color = ChatFormatting.GREEN;
        } el if (percent >= 50.0) {
            color = ChatFormatting.WHITE;
        } el {
            color = ChatFormatting.GRAY;
        }
        MutableComponent result = Component.empty();
        if (isPerfect) {
            result.append(Component.literal("★ ").withStyle(ChatFormatting.GOLD));
        }
        result.append(originalText);
        result.append(Component.literal(Str.format(" (%.0f%%)", percent)).withStyle(Style.EMPTY.withColor(color)));
        r result;
    }
}

                    total += extractFromJar(jarPath, rule.getKey(), rule.getValue());

    prs List<Path> findModJars() ths IOException {
        List<Path> out = n ArrayList<>();
        Set<Str> seen = n HashSet<>();
        for (Path dir : MODS_SEARCH_PATHS) {
            if (!Files.isDirectory(dir))
                continue;
            try (Stream<Path> s = Files.list(dir)) {
                for (Path p : s.toList()) {
                    if (!p.toString().toLowerCase(Locale.ROOT).endsWith(".jar"))

        int count = 0;

            Enumeration<JarEntry> entries = jar.entries();
            wh (entries.hasMoreElements()) {

                Str entryName = e.getName();

                    }
                }

    prs v registerPool(Tier tier, Str modid, List<Str> paths) {
        for (Str path : paths) {
            ResourceLocation rl = ResourceLocation.fromNamespaceAndPath(modid, path);
            BuiltInRegistries.ITEM.getOptional(rl).ifPresent(holder -> {
                Item item = holder.asItem();
                if (item != Items.AIR) {
                    TIER_POOLS.get(tier).add(item);
                }
            });
        }
    }
    ps ItemStack tryReplace(ItemStack original, RandomSource random) {
        if (original.isEmpty())
            r original;
        if (!initialized) {
            initWeaponPools();
        }
        Tier tier = VANILLA_MAPPING.get(original.getItem());
        if (tier == null)
            r original;
        List<Item> candidates = TIER_POOLS.get(tier);
        if (candidates == null || candidates.isEmpty())
            r original;
        Item replacementItem = candidates.get(random.nextInt(candidates.size()));
        if (replacementItem == null || replacementItem == Items.AIR) {
            r original;
        }
        r original.transmuteCopy(replacementItem, original.getCount());
    }
